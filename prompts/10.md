# 2. Corrigir ordem de persistência, `trusted_chain_head` e `last_accepted_epoch`

Problema atual:

`takeSnapshot` pode fazer:

```zig
self.trusted_chain_head = snapshot_id;
```

antes de persistir e confirmar receipt.

Isso é incorreto. `takeSnapshot` ou `buildPendingSnapshot` deve construir o envelope, mas não pode mutar o estado criptográfico global.

Crie tipos explícitos:

```zig
pub const PendingSnapshot = struct {
    envelope: crypto.SnapshotEnvelope,
    snapshot_id: [32]u8,
    epoch: u64,
};

pub const CommittedSnapshot = struct {
    envelope: crypto.SnapshotEnvelope,
    snapshot_id: [32]u8,
    epoch: u64,
    persistence_receipt: crypto.PersistenceReceipt,
};
```

```zig

    ctx: *crypto.VerificationContext,

    const next_epoch = self.last_accepted_epoch + 1;

    // build plaintext state
    // generate DEK
    // seal payload
    // wrap DEK
    // build snapshot_body
    // compute snapshot_id
    // create attestations
    // return PendingSnapshot
    // DO NOT update trusted_chain_head
    // DO NOT update last_accepted_epoch

```

Depois implemente o fluxo de commit:

```zig
fn persistEnvelopeLocal(
    self: *Storage,
    allocator: std.mem.Allocator,
    envelope_without_receipt: crypto.SnapshotEnvelope,
) !CommittedSnapshot {
    // 1. save envelope without receipt or to temp file
    try self.saveEnvelopeJson(allocator, envelope_without_receipt);

    // 2. create receipt
    const receipt = crypto.PersistenceReceipt{
        .provider = "local",
        .repository = "local-dev",
        .branch = "main",
        .commit_sha = null,
        .committed_at = "2026-05-18T00:00:00Z",
    };

    // 3. attach receipt
    var final_envelope = envelope_without_receipt;
    final_envelope.persistence_receipt = receipt;

    // 4. save final envelope
    try self.saveEnvelopeJson(allocator, final_envelope);

    return .{
        .envelope = final_envelope,
        .snapshot_id = final_envelope.snapshot_id,
        .epoch = final_envelope.snapshot_body.epoch,
        .persistence_receipt = receipt,
    };
}
```

`writeSecret` deve ser o único lugar que avança o estado depois da persistência confirmada:

```zig

    };
```

# 3. Corrigir `NonceRegistry` para ser idempotente por `snapshot_id`

Problema atual:

Se o mesmo snapshot for validado duas vezes, pode dar falso `NonceReused`.

A validação de nonce deve ser idempotente para o mesmo `snapshot_id`, mas deve rejeitar reuse em outro snapshot.

Implemente:

```zig
pub const NonceUseKind = enum {
    payload,
    dek_wrap,
};

pub const NonceRecord = struct {
    snapshot_id: [32]u8,
    kind: NonceUseKind,
    nonce: [12]u8,
    context_hash: [32]u8,
};

pub const NonceRegistry = struct {
    seen: std.AutoHashMap([32]u8, NonceRecord),

    pub fn init(allocator: std.mem.Allocator) NonceRegistry {
        return .{
            .seen = std.AutoHashMap([32]u8, NonceRecord).init(allocator),
        };
    }

    pub fn deinit(self: *NonceRegistry) void {
        self.seen.deinit();
    }

    pub fn checkAndRecord(
        self: *NonceRegistry,
        snapshot_id: [32]u8,
        kind: NonceUseKind,
        context: []const u8,
        nonce: [12]u8,
    ) !void {
        const key = computeNonceKey(kind, context, nonce);

        if (self.seen.get(key)) |existing| {
            if (std.mem.eql(u8, &existing.snapshot_id, &snapshot_id)) {
                // same snapshot, same nonce/context: idempotent verification
                return;
            }

            return LsesError.NonceReused;
        }

        try self.seen.put(key, .{
            .snapshot_id = snapshot_id,
            .kind = kind,
            .nonce = nonce,
            .context_hash = sha256Bytes(context),
        });
    }
};
```

Implemente:

```zig
pub fn computeNonceKey(
    kind: NonceUseKind,
    context: []const u8,
    nonce: [12]u8,
) [32]u8 {
    var h = std.crypto.hash.sha2.Sha256.init(.{});
    h.update("LSES-NONCE-KEY-V1");

    switch (kind) {
        .payload => h.update("payload"),
        .dek_wrap => h.update("dek_wrap"),
    }

    h.update("\x00");
    h.update(context);
    h.update("\x00");
    h.update(&nonce);

    var out: [32]u8 = undefined;
    h.final(&out);
    return out;
}
```

# 5. Corrigir `recoverLatest` para validação sequencial de chain

Problema atual:

`recoverLatest` pode chamar `verifySnapshot` para cada snapshot usando o mesmo `trusted_chain_head`, o que não é correto para validar uma cadeia histórica.

A validação correta precisa ser sequencial:

* snapshot 1 deve apontar para anchor inicial
* snapshot 2 deve apontar para hash do snapshot 1
* snapshot 3 deve apontar para hash do snapshot 2
* etc.

Implemente função auxiliar:

```zig
fn sortByEpoch(envelopes: []crypto.SnapshotEnvelope) void {
    std.mem.sort(
        crypto.SnapshotEnvelope,
        envelopes,
        {},
        struct {
            fn lessThan(_: void, a: crypto.SnapshotEnvelope, b: crypto.SnapshotEnvelope) bool {
                return a.snapshot_body.epoch < b.snapshot_body.epoch;
            }
        }.lessThan,
    );
}
```

`recoverLatest` deve fazer:

```zig
pub fn recoverLatest(
    self: *Storage,
    allocator: std.mem.Allocator,
    envelopes: []crypto.SnapshotEnvelope,
    ctx: *crypto.VerificationContext,
) !RecoveryResult {
    if (envelopes.len == 0) return crypto.LsesError.InvalidEnvelope;

    var sorted = try allocator.dupe(crypto.SnapshotEnvelope, envelopes);
    defer allocator.free(sorted);

    sortByEpoch(sorted);

    var limited = false;

    var expected_previous: [32]u8 = undefined;

    if (ctx.trusted_chain_head) |head| {
        expected_previous = head;
    } else {
        if (ctx.mode == .strict) return crypto.LsesError.MissingTrustAnchor;
        limited = true;
        expected_previous = sorted[0].snapshot_body.previous_snapshot_hash;
    }

    var last_epoch: u64 = ctx.last_accepted_epoch;

    for (sorted) |env| {
        if (env.snapshot_body.epoch <= last_epoch) {
            return crypto.LsesError.RollbackDetected;
        }

        if (!std.mem.eql(u8, &env.snapshot_body.previous_snapshot_hash, &expected_previous)) {
            return crypto.LsesError.ChainHeadMismatch;
        }

        var per_snapshot_ctx = ctx.*;
        per_snapshot_ctx.trusted_chain_head = expected_previous;
        per_snapshot_ctx.last_accepted_epoch = last_epoch;

        try crypto.verifySnapshot(allocator, env, &per_snapshot_ctx);

        expected_previous = env.snapshot_id;
        last_epoch = env.snapshot_body.epoch;
    }

    const latest = sorted[sorted.len - 1];

    const plaintext = try crypto.decryptVerifiedSnapshot(
        allocator,
        latest,
        ctx,
    );
    defer {
        crypto.secureZero(plaintext);
        allocator.free(plaintext);
    }

    const handle = try self.runtime_store.insert(
        allocator,
        plaintext,
        60,
    );

    return .{
        .latest_snapshot_id = latest.snapshot_id,
        .latest_epoch = latest.snapshot_body.epoch,
        .limited = limited,
        .restored_handle = handle,
    };
}
```

Ajuste se `decryptVerifiedSnapshot` revalidar com head antigo. Uma opção melhor é criar:

```zig
decryptVerifiedSnapshotWithContext(...)
```

usando o contexto correto para o último snapshot.

```zig
            .secret = copy,
            .expires_at = expires_at,
            .consumed = false,
        });

        return .{
            .handle_id = handle_id,
            .expires_at = expires_at,
            .scope = "runtime-only",
        };
    }

    pub fn consume(
        self: *RuntimeSecretStore,
        allocator: std.mem.Allocator,
        handle_id: [32]u8,
    ) ![]u8 {
        var entry = self.entries.getPtr(handle_id) orelse return crypto.LsesError.InvalidHandle;

        if (entry.consumed or std.time.timestamp() > entry.expires_at) {
            crypto.secureZero(entry.secret);
            allocator.free(entry.secret);
            _ = self.entries.remove(handle_id);
            return crypto.LsesError.ExpiredHandle;
        }

        const out = try allocator.dupe(u8, entry.secret);

        crypto.secureZero(entry.secret);
        allocator.free(entry.secret);
        entry.consumed = true;
        _ = self.entries.remove(handle_id);

        return out;
    }

    pub fn purgeExpired(
        self: *RuntimeSecretStore,
        allocator: std.mem.Allocator,
    ) void {
        // iterate; if expired, secureZero + free + remove
    }
};
```

```md
The in-memory `NonceRegistry` is idempotent for repeated verification of the same `snapshot_id`, but rejects nonce reuse across different snapshots in the same nonce domain.
```
